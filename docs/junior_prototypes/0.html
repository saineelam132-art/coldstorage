Lambda: check-in (Python)

import json, os, uuid, hmac, hashlib, boto3, decimal
ddb = boto3.resource("dynamodb").Table(os.environ["TABLE"]); KEY = os.environ["QR_KEY"].encode()
def handler(event, ctx):
    b = json.loads(event["body"]); q = float(b["quantityKg"])
    if q <= 0 or q > 150: return {"statusCode":400,"body":"Max 150 kg"}
    slip = "CT-" + uuid.uuid4().hex[:8].upper(); bay = "B" + str(int(slip[3:5],16) % 6 + 1)
    gen = hashlib.sha256(f"0|{slip}|{b['farmerName']}|{q}".encode()).hexdigest()
    sig = hmac.new(KEY, f"{slip}|{bay}|{gen}".encode(), hashlib.sha256).hexdigest()
    ddb.put_item(Item={"slipId":slip,"bay":bay,"farmer":b["farmerName"],"produce":b["produceType"],
        "qtyKg":decimal.Decimal(str(q)),"basePrice":decimal.Decimal(str(b["basePrice"])),"genesisHash":gen,"status":"STORED"})
    return {"statusCode":200,"body":json.dumps({"slipId":slip,"bay":bay,"qr":{"slipId":slip,"bay":bay,"genesisHash":gen,"sig":sig}})}

Lambda: buyer verify & settle (Python)

def verify(chain):
    prev = "0"*64
    for b in chain:
        if b["prevHash"] != prev or h(b["index"],b["ts"],b["slipId"],b["payload"],b["prevHash"]) != b["hash"]:
            return False, b["index"]
        prev = b["hash"]
    return True, None
def freshness(chain):
    f = 100.0
    for b in chain[1:]:
        t = b["payload"]["nodes"]["n2"]["coreTempC"]; dev = max(2-t, t-8, 0)
        f -= 0.10 + 0.40 * dev ** 1.6
    return max(f, 0)
def handler(event, ctx):
    chain = load_chain(event["slipId"]); ok, bad = verify(chain)
    if not ok:
        freeze_payment(event["slipId"]); return {"alert":"EVIDENCE TAMPERED","brokenAt":bad,"mode":"PENALTY"}
    f = freshness(chain); return {"freshness":f,"finalPrice":base_price(event["slipId"]) * f/100}
